# The €10K Site Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Lift `/cinematic/` (EN + DE) from "€10K hero, €200 body" to €10K on all eight checklist points, and sell that build as a `cinematic-site` package on `/pricing/`.

**Architecture:** Everything is repo-owned and applied at deploy time. Case-study content lives in `seo/case-studies.json` and is rendered into both cinematic homepages by a new `seo/apply_cases.py` layer (same gate pattern as `seo/apply_legal.py`: exit 2 and change nothing while drafts exist). Static markup changes (pillars, demo, mobile, pricing tier) are edited directly in `pages/cinematic/**` and `pages/pricing/`, which `apply_pages` copies verbatim. A stdlib `tests/test_cinematic.py` guards the new content the way `tests/test_pricing_palette.py` guards the pricing page.

**Tech Stack:** bash (`deploy.sh`), Python 3 stdlib + Pillow 11 (avif/webp enabled), vanilla CSS/JS in `pages/cinematic/`, ffmpeg, nginx, Stripe Payment Links, n8n. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-09-13-10k-site-design.md`

## Global Constraints

- Never edit `site/` or `frontend/template/` — they are wiped every deploy. Sources: `pages/`, `seo/`, `images/`, `brand/`, `deploy/`.
- EN (`pages/cinematic/index.html`) and DE (`pages/cinematic/de/index.html`) change together in every task that touches the homepage.
- Coral `#fb5e31` = action only (buttons, links, focus, wordmark). Teal/emerald = imagery only. No emerald text or borders remain in the UI.
- Type scale: Metric `clamp(56px,8vw,104px)` 800 `tabular-nums` −0.03em; H2 `clamp(36px,4.5vw,60px)` 700 −0.02em; H3 24px/1.25; body 17px/1.6 `max-width:62ch`; kicker mono 12px .18em uppercase muted. Section rhythm `--sec: clamp(96px,12vw,176px)`.
- Nothing fake ships: `apply_cases.py` exits 2 and writes nothing when any case has `"draft": true` unless `GEORGEAI_ENV=dev`.
- All motion gated by `prefers-reduced-motion` (the page already has `rmq` in `script.js`).
- Tests are stdlib-only, exit 0/1, runnable as `python3 tests/<file>.py` (same contract as `tests/test_payments_pipeline.py`).
- Prices live in three places that must match: `pages/pricing/index.html`, Stripe, `CATALOG` in `tests/test_payments_pipeline.py`. The 10K package's Stripe link is a **€2,500 deposit**: `CATALOG["cinematic-site"] = {"one_time": 250000, "monthly": None}`.
- Commit messages: conventional, end with `Co-Authored-By: Claude Opus 5 (1M context) <noreply@anthropic.com>`.
- Work on a branch cut from `main`: `git checkout main && git pull && git checkout -b feat/10k-site`.

`seo/apply_cases.py`:
```python
#!/usr/bin/env python3
"""Render seo/case-studies.json into both cinematic homepages.

Usage: apply_cases.py <site_dir> <seo_dir>
Exit 0 applied · 2 drafts present in production (nothing written) · 1 error.

Idempotent full replace between <!-- cases:start --> and <!-- cases:end -->.
Set GEORGEAI_ENV=dev to render drafts locally (they get a visible DRAFT ribbon).
Images are referenced as /cinematic/assets/cases/<stem>.{avif,webp,jpg};
images/gen_cases.py (Task 6) produces them — until then the <picture> falls
back to the jpg copied by that same script.
"""
from __future__ import annotations

import html
import json
import os
import sys
from pathlib import Path

START, END = "<!-- cases:start -->", "<!-- cases:end -->"
PAGES = {"en": "cinematic/index.html", "de": "cinematic/de/index.html"}
T = {
    "en": {"kicker": "Proof", "cases": "Case", "h2": "Three systems, three numbers.", "sub": "Real clients, real screens, real periods. Nothing on this page is a mock.",
           "problem": "Problem", "system": "System", "result": "Result", "draft": "DRAFT — not yet verified"},
    "de": {"kicker": "Belege", "cases": "Fall", "h2": "Drei Systeme, drei Zahlen.", "sub": "Echte Kunden, echte Bildschirme, echte Zeiträume. Nichts auf dieser Seite ist ein Mockup.",
           "problem": "Problem", "system": "System", "result": "Ergebnis", "draft": "ENTWURF — noch nicht verifiziert"},
}


def esc(s: str) -> str:
    return html.escape(s, quote=True)


def picture(c: dict, lang: str) -> str:
    stem = Path(c["img"]).stem
    base = f"/cinematic/assets/cases/{stem}"
    return (f'<picture><source type="image/avif" srcset="{base}.avif"><source type="image/webp" srcset="{base}.webp">'
            f'<img src="{base}.jpg" alt="{esc(c["img_alt"][lang])}" loading="lazy" width="1600" height="1000"></picture>')


def render(cases: list[dict], lang: str) -> str:
    t = T[lang]
    strip = "".join(
        f'<li><b class="num" data-to="{esc(c["metric"])}" data-prefix="{esc(c.get("metric_prefix", ""))}" data-suffix="{esc(c.get("metric_suffix", ""))}">0</b>'
        f'<span>{esc(c["metric_label"][lang])}</span><small class="mono">{esc(c["client"])}</small></li>'
        for c in cases)
    studies = "".join(
        f'<article class="case reveal{" case-draft" if c.get("draft") else ""}" id="case-{esc(c["slug"])}">'
        f'{"<p class=\"ribbon mono\">" + t["draft"] + "</p>" if c.get("draft") else ""}'
        f'<figure class="case-media">{picture(c, lang)}</figure>'
        f'<div class="case-text"><p class="kicker mono">{i:02d} / {t["cases"]}</p>'
        f'<p class="metric"><b class="num" data-to="{esc(c["metric"])}" data-prefix="{esc(c.get("metric_prefix", ""))}" data-suffix="{esc(c.get("metric_suffix", ""))}">0</b>'
        f'<span>{esc(c["metric_label"][lang])}</span></p><h3>{esc(c["client"])}</h3>'
        f'<dl><dt>{t["problem"]}</dt><dd>{esc(c["problem"][lang])}</dd><dt>{t["system"]}</dt><dd>{esc(c["system"][lang])}</dd>'
        f'<dt>{t["result"]}</dt><dd>{esc(c["result"][lang])}</dd></dl></div></article>'
        for i, c in enumerate(cases, 1))
    return (f'<section class="proof" aria-label="{t["kicker"]}"><ul class="stat-track proof-track">{strip}</ul></section>\n'
            f'<section class="cases" id="work" aria-labelledby="cases-h"><header class="sec-head reveal"><p class="kicker mono">{t["kicker"]}</p>'
            f'<h2 id="cases-h">{t["h2"]}</h2><p>{t["sub"]}</p></header>{studies}</section>')


def main() -> int:
    site, seo = Path(sys.argv[1]), Path(sys.argv[2])
    data = json.loads((seo / "case-studies.json").read_text())
    cases = data["cases"]
    if any(c.get("draft") for c in cases) and os.environ.get("GEORGEAI_ENV") != "dev":
        print("apply_cases: draft case studies present — nothing written", file=sys.stderr)
        return 2
    for lang, rel in PAGES.items():
        page = site / rel
        if not page.exists():
            continue
        src = page.read_text()
        a, b = src.find(START), src.find(END)
        if a < 0 or b < 0:
            print(f"apply_cases: markers missing in {rel}", file=sys.stderr)
            return 1
        out = src[: a + len(START)] + "\n" + render(cases, lang) + "\n" + src[b:]
        if out != src:
            page.write_text(out)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

- [ ] **Step 1: gen_cases.py**

```python
#!/usr/bin/env python3
"""images/cases/*.{jpg,png} → site/cinematic/assets/cases/<stem>.{avif,webp,jpg} at 1600px wide.
Usage: gen_cases.py <site_dir>. Skips outputs newer than their source. Pillow 11 (avif+webp built in)."""
import sys
from pathlib import Path
from PIL import Image

SRC = Path(__file__).resolve().parent / "cases"
W = 1600


def main() -> int:
    out = Path(sys.argv[1]) / "cinematic" / "assets" / "cases"
    out.mkdir(parents=True, exist_ok=True)
    for src in sorted(p for p in SRC.iterdir() if p.suffix.lower() in (".jpg", ".jpeg", ".png")):
        targets = {ext: out / f"{src.stem}{ext}" for ext in (".avif", ".webp", ".jpg")}
        if all(t.exists() and t.stat().st_mtime >= src.stat().st_mtime for t in targets.values()):
            continue
        im = Image.open(src).convert("RGB")
        if im.width > W:
            im = im.resize((W, round(im.height * W / im.width)), Image.LANCZOS)
        im.save(targets[".avif"], quality=55)
        im.save(targets[".webp"], quality=78, method=6)
        im.save(targets[".jpg"], quality=82, optimize=True, progressive=True)
        print("cases:", src.name, "→", ", ".join(f"{t.suffix[1:]} {t.stat().st_size // 1024} KB" for t in targets.values()))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
